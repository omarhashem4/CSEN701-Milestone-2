cmake_minimum_required(VERSION 3.13)

# Use the tools installed by the official Raspberry Pi Pico VS Code extension.
set(USERHOME $ENV{HOME})
set(sdkVersion 2.3.1)
set(toolchainVersion 15_2_Rel1)
set(picotoolVersion 2.3.1)
set(picoVscode "${USERHOME}/.pico-sdk/cmake/pico-vscode.cmake")
if(EXISTS "${picoVscode}")
    include("${picoVscode}")
endif()

set(PICO_BOARD pico CACHE STRING "Pico board")
include(pico_sdk_import.cmake)
project(musical_orientation C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()


function(component_test name)
    add_executable(${name} ${ARGN})
    target_include_directories(${name} PRIVATE drivers)
    target_link_libraries(${name} pico_stdlib hardware_i2c hardware_pwm hardware_clocks)
    pico_enable_stdio_usb(${name} 1)
    pico_enable_stdio_uart(${name} 0)
    pico_add_extra_outputs(${name})
endfunction()

component_test(touch tests/touch.c drivers/ttp223.c)
component_test(motion tests/motion.c drivers/mpu6050.c)
component_test(lcd tests/lcd.c drivers/lcd.c)

component_test(servo_gp7 tests/servo.c drivers/servo.c)

target_compile_definitions(servo_gp8 PRIVATE SERVO_PIN=8)

component_test(buzzer tests/buzzer.c drivers/buzzer.c)
